PHASE 2: AUTOMATED GGPLOT2 THUMBNAILS
--------------------------------------
Time estimate: 1 hour

Create R script: scripts/generate_ggplot_thumbnails.R

```r
library(ggplot2)
library(dplyr)

# EKIO brand colors
ekio_primary <- "#1abc9c"
ekio_secondary <- "#16a085"

# Standard theme for all thumbnails
theme_ekio_thumbnail <- function() {
  theme_minimal(base_size = 14) +
    theme(
      plot.background = element_rect(fill = "white", color = NA),
      panel.background = element_rect(fill = "#f8f9fa", color = NA),
      plot.margin = margin(20, 20, 20, 20)
    )
}

# Save function with consistent specs
save_thumbnail <- function(plot, filename) {
  ggsave(
    filename,
    plot,
    width = 8,
    height = 5,
    dpi = 150,
    bg = "white"
  )
}

# Generate scatter plot
p_scatter <- ggplot(mtcars, aes(wt, mpg)) +
  geom_point(color = ekio_primary, size = 3, alpha = 0.7) +
  geom_smooth(method = "lm", se = FALSE, color = ekio_secondary) +
  labs(title = "Gráfico de Dispersão",
       x = "Peso", y = "Consumo") +
  theme_ekio_thumbnail()

save_thumbnail(p_scatter, "static/ggplot2_point.png")

# Generate histogram
p_hist <- ggplot(mtcars, aes(mpg)) +
  geom_histogram(fill = ekio_primary, color = "white", bins = 10) +
  labs(title = "Histograma", x = "Consumo", y = "Frequência") +
  theme_ekio_thumbnail()

save_thumbnail(p_hist, "static/ggplot2_histogram.png")

# Generate bar plot
diamonds_summary <- diamonds %>%
  count(cut) %>%
  mutate(cut = forcats::fct_reorder(cut, n))

p_bar <- ggplot(diamonds_summary, aes(cut, n)) +
  geom_col(fill = ekio_primary, width = 0.7) +
  coord_flip() +
  labs(title = "Gráfico de Barras", x = NULL, y = "Contagem") +
  theme_ekio_thumbnail()

save_thumbnail(p_bar, "static/ggplot2_bar.png")

# Generate line plot
p_line <- ggplot(economics, aes(date, unemploy)) +
  geom_line(color = ekio_primary, linewidth = 1) +
  labs(title = "Gráfico de Linha",
       x = NULL, y = "Desemprego (milhares)") +
  theme_ekio_thumbnail()

save_thumbnail(p_line, "static/ggplot2_line.png")

cat("✓ Generated 4 ggplot2 thumbnails\n")
```

Run with: Rscript scripts/generate_ggplot_thumbnails.R
